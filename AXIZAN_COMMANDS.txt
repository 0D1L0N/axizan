# Axizán - Commandes utiles

## 1) Pré-requis

Vérifiez que ces outils sont installés :

php --version
node --version
composer --version
psql --version

## 2) Installation initiale du projet

# À la racine du projet
chmod +x install.sh
./install.sh

# Si vous préférez faire étape par étape :
cd backend
composer install
cp .env.example .env
php artisan key:generate

cd ../frontend
npm install
cp .env.example .env.local

## 3) Base de données PostgreSQL

# Démarrer PostgreSQL (Linux)
sudo systemctl start postgresql

# Créer la base de données
createdb axizan

# Ou via psql
psql -U postgres
CREATE DATABASE axizan;
\q

## 4) Configuration des fichiers d'environnement

# Backend
cd backend
cp .env.example .env
# Modifier backend/.env si nécessaire

# Frontend
cd ../frontend
cp .env.example .env.local
# Modifier frontend/.env.local si nécessaire

## 5) Backend Laravel

cd backend
php artisan migrate
php artisan db:seed
php artisan serve

# Autres commandes Laravel utiles
php artisan migrate:rollback
php artisan migrate:fresh --seed
php artisan tinker
php artisan test
php artisan cache:clear
php artisan config:cache

## 6) Frontend Next.js

cd frontend
npm install
npm run dev
npm run build
npm start
npm run lint
npm test
npm run test:watch

## 7) Commandes utilitaires du script dev.sh

# Depuis la racine du projet
chmod +x dev.sh
./dev.sh serve        # Lancer backend (port 8000)
./dev.sh migrate      # Exécuter les migrations
./dev.sh seed         # Exécuter les seeders
./dev.sh fresh        # Réinitialiser la BD + seed
./dev.sh tinker       # PHP REPL
./dev.sh test         # Tests backend
./dev.sh log          # Suivre le log Laravel
./dev.sh dev          # Lancer frontend (port 3000)
./dev.sh build        # Build frontend production
./dev.sh lint         # Vérifier le frontend
./dev.sh test:fe      # Tests frontend
./dev.sh check        # Vérifier les prérequis
./dev.sh clean        # Nettoyer le cache Laravel
./dev.sh db:create    # Créer la base axizan
./dev.sh db:drop      # Supprimer la base axizan

## 8) Commandes Docker / docker-compose

docker-compose up -d
docker-compose down

docker-compose exec laravel php artisan migrate
docker-compose exec laravel php artisan db:seed

# Vérifier les services
docker ps

## 9) URL par défaut

Frontend : http://localhost:3000
Backend : http://localhost:8000
Mailpit : http://localhost:8025

curl -X POST http://localhost:8000/api/v1/auth/login \

curl http://localhost:8000/api/v1/products

## 11) Vérifications courantes

which php
which node
which composer
php -m | grep pdo_pgsql

# Redémarrer PostgreSQL
sudo systemctl restart postgresql

# Si le port 3000 est déjà utilisé
npm run dev -- -p 3001

## 12) Fichiers importants

backend/.env
frontend/.env.local
backend/composer.json
frontend/package.json
docker-compose.yml
install.sh
dev.sh

## 13) Notes

- Lancer le backend avant de lancer le frontend.
- Si vous utilisez Docker, la base et Redis sont fournis par docker-compose.
- Si la connexion API échoue, vérifiez que le backend tourne et que l'URL API est correcte dans frontend/.env.local.
